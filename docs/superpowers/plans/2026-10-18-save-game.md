# Saving the Game and Stocking Ready Boards — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A reload never loses the game in progress, and every new board (first visit included, any rope count) appears instantly.

**Architecture:** Snapshots keep rope ids (so colors follow from the seed). A `Stock` mirrors one ready board per rope count in localStorage; `BoardSource` takes from it and refills it one generator job at a time. Pre-generated boards in the repo fill an empty stock. A `GameSaver` writes the current game whenever the session comes to rest; the app resumes it at startup.

**Tech Stack:** TypeScript (strict), Vite 8, Vitest (globals, node environment), Playwright e2e, oxlint. Node 22 (`.nvmrc`).

**Spec:** `docs/superpowers/specs/2026-10-18-save-game-design.md`

## Review Focus

1. Storage that refuses writes (quota full, private mode): the stock keeps boards in memory and the game is playable, just unsaved — Task 3 (stock) and Task 8 (saver) test a throwing store.
2. Bundled chunks that fail to load (first visit while offline, blocked request): the first board falls back to the generator — Task 6 tests a rejecting loader; Task 7 makes the generator-failure e2e tests block bundled chunks.
3. A generator result that someone is waiting for must not also go into the stock (it would hand out the same board twice) — Task 4 tests it.
4. A reload while a replacement board is still being made resumes the old game, and a hide during that wait doesn't rewrite it — Task 8 tests `forget()`.
5. A page hidden while a rope is held or flying writes nothing (the snapshot would throw or capture a mid-move state) — Task 8 tests it.

```ts
import { PITCH, rimBoard } from "../board";
import { Engine } from "./engine";
import type { P3 } from "./types";

export type RopeSnapshot = { pts: Float64Array<ArrayBuffer>; ends: [number, number] } | { cleared: true };

export interface EngineSnapshot {
  cols: number;
  rows: number;
  ropes: RopeSnapshot[];
}

export function toSnapshot(E: Engine): EngineSnapshot {
  return {
    cols: Math.round(E.board.width / PITCH) + 1,
    rows: Math.round(E.board.height / PITCH) + 1,
    ropes: E.ropes.map((r): RopeSnapshot => {
      if (r.cleared) return { cleared: true };
      const [a, b] = r.ends;
      if (a === null || b === null) throw new Error(`toSnapshot: rope ${r.id} has an end out of its hole`);
      const pts = new Float64Array(r.pts.length * 6);
      r.pts.forEach((p, i) => pts.set([p.x, p.y, p.z, p.px, p.py, p.pz], i * 6));
      return { pts, ends: [a, b] };
    }),
  };
}

export function fromSnapshot(s: EngineSnapshot): Engine {
  const E = new Engine(rimBoard(s.cols, s.rows));
  for (const r of s.ropes) {
    if ("cleared" in r) {
      E.addClearedRope();
      continue;
    }
    const pts: P3[] = [];
    for (let i = 0; i < r.pts.length; i += 6) pts.push({ x: r.pts[i]!, y: r.pts[i + 1]!, z: r.pts[i + 2]! });
    const rope = E.addRope(pts, r.ends);
    rope.pts.forEach((p, i) => {
      p.px = r.pts[i * 6 + 3]!;
      p.py = r.pts[i * 6 + 4]!;
      p.pz = r.pts[i * 6 + 5]!;
    });
  }
  return E;
}

export function snapshotTransfer(s: EngineSnapshot): ArrayBuffer[] {
  return s.ropes.flatMap((r) => ("cleared" in r ? [] : [r.pts.buffer]));
}
```

```ts
import { boardForRopes, PITCH } from "../board";
import type { EngineSnapshot, RopeSnapshot } from "../engine/snapshot";
import type { KeyValueStore } from "./settings";

export const RECORD_VERSION = 1;
export const ROPE_COUNTS: readonly number[] = [4, 5, 6, 7, 8, 9, 10];

export interface StorageKeys {
  game: string;
  ready(ropes: number): string;
}

export function storageKeys(base: string): StorageKeys {
  const prefix = `rope-tangle${base}`;
  return { game: `${prefix}game`, ready: (ropes) => `${prefix}ready/${ropes}` };
}

export interface ReadyBoard {
  ropes: number;
  seed: number;
  generationMs: number;
  tries: number;
  snapshot: EngineSnapshot;
}

type Obj = Record<string, unknown>;

const isObj = (v: unknown): v is Obj => typeof v === "object" && v !== null && !Array.isArray(v);
const isNum = (v: unknown): v is number => typeof v === "number" && Number.isFinite(v);
const isInt = (v: unknown): v is number => Number.isInteger(v);

export function encodeSnapshot(s: EngineSnapshot): unknown {
  return {
    cols: s.cols,
    rows: s.rows,
    ropes: s.ropes.map((r) => ("cleared" in r ? { cleared: true } : { pts: Array.from(r.pts), ends: [...r.ends] })),
  };
}

export function decodeSnapshot(v: unknown, ropes: number): EngineSnapshot | null {
  if (!ROPE_COUNTS.includes(ropes) || !isObj(v) || !Array.isArray(v["ropes"]) || v["ropes"].length !== ropes) return null;
  const board = boardForRopes(ropes);
  const cols = Math.round(board.width / PITCH) + 1, rows = Math.round(board.height / PITCH) + 1;
  if (v["cols"] !== cols || v["rows"] !== rows) return null;
  const used = new Set<number>(), out: RopeSnapshot[] = [];
  for (const r of v["ropes"]) {
    if (!isObj(r)) return null;
    if (r["cleared"] === true) {
      out.push({ cleared: true });
      continue;
    }
    const pts = r["pts"], ends = r["ends"];
    if (!Array.isArray(pts) || pts.length < 12 || pts.length % 6 !== 0 || !pts.every(isNum)) return null;
    if (!Array.isArray(ends) || ends.length !== 2) return null;
    for (const h of ends) {
      if (!isInt(h) || h < 0 || h >= board.holes.length || used.has(h)) return null;
      used.add(h);
    }
    out.push({ pts: Float64Array.from(pts), ends: [ends[0] as number, ends[1] as number] });
  }
  return { cols, rows, ropes: out };
}

export function encodeReady(b: ReadyBoard): unknown {
  return { v: RECORD_VERSION, ropes: b.ropes, seed: b.seed, generationMs: b.generationMs, tries: b.tries, snapshot: encodeSnapshot(b.snapshot) };
}

export function decodeReady(v: unknown, ropes: number): ReadyBoard | null {
  if (!isObj(v) || v["v"] !== RECORD_VERSION || v["ropes"] !== ropes) return null;
  const { seed, generationMs, tries } = v;
  if (!isInt(seed) || !isNum(generationMs) || !isInt(tries)) return null;
  const snapshot = decodeSnapshot(v["snapshot"], ropes);
  return snapshot ? { ropes, seed, generationMs, tries, snapshot } : null;
}

export function removeRecord(store: KeyValueStore, key: string): void {
  try {
    store.removeItem(key);
  } catch {
    return;
  }
}

export function readRecord<T>(store: KeyValueStore, key: string, decode: (v: unknown) => T | null): T | null {
  let raw: string | null;
  try {
    raw = store.getItem(key);
  } catch {
    return null;
  }
  if (raw === null) return null;
  let value: T | null = null;
  try {
    value = decode(JSON.parse(raw));
  } catch {
    value = null;
  }
  if (value === null) removeRecord(store, key);
  return value;
}

export function writeRecord(store: KeyValueStore, key: string, value: unknown): boolean {
  try {
    store.setItem(key, JSON.stringify(value));
    return true;
  } catch {
    return false;
  }
}
```

```ts
import { decodeReady, encodeReady, readRecord, removeRecord, ROPE_COUNTS, writeRecord, type ReadyBoard, type StorageKeys } from "./records";
import type { KeyValueStore } from "./settings";

export class Stock {
  private readonly boards = new Map<number, ReadyBoard>();
  private readonly store: KeyValueStore;
  private readonly keys: StorageKeys;

  constructor(store: KeyValueStore, keys: StorageKeys) {
    this.store = store;
    this.keys = keys;
    for (const n of ROPE_COUNTS) {
      const b = readRecord(store, keys.ready(n), (v) => decodeReady(v, n));
      if (b) this.boards.set(n, b);
    }
  }

  has(ropes: number): boolean {
    return this.boards.has(ropes);
  }

  take(ropes: number): ReadyBoard | null {
    const b = this.boards.get(ropes);
    if (!b) return null;
    this.boards.delete(ropes);
    removeRecord(this.store, this.keys.ready(ropes));
    return b;
  }

  put(b: ReadyBoard): void {
    this.boards.set(b.ropes, b);
    writeRecord(this.store, this.keys.ready(b.ropes), encodeReady(b));
```

```ts
  private obtain(ropes: number): Promise<LoadedBoard> {

    if (j && j.ropes === ropes && this.pending.has(j.id)) {
      const prefetched = j.background && !j.claimed;
      j.claimed = true;
      return j.promise.then((b) => ({ ...b, prefetched }));
    }
    return this.start(ropes, false).promise;
  }

  private pump(): void {
    if (!this.filling || (this.job && this.pending.has(this.job.id))) return;
    const missing = ROPE_COUNTS.filter((n) => !this.stock.has(n) && !this.failed.has(n));
    const n = fillOrder(this.current, missing)[0];

      try {
        const worker = (this.worker ??= this.spawn());
        this.pending.set(id, { ropes, timer: this.watch(id, ropes), resolve, reject });
        worker.postMessage({ id, seed: randomSeed(), ropes }, { transfer: [] });

        reject(new BoardFailure("crashed", e instanceof Error ? e.message : String(e)));
      }

          if (!job.claimed && !(e instanceof BoardFailure && e.kind === "superseded")) this.failed.add(ropes);
        },
      )
      .finally(() => {
        if (this.job === job) this.job = null;
        queueMicrotask(() => this.pump());
      });
```

```ts
import { decodeReady, ROPE_COUNTS } from "./records";
import type { Stock } from "./stock";

export type BundledLoader = (ropes: number) => Promise<unknown>;

const files = import.meta.glob<unknown>("./bundled/ready-*.json", { import: "default" });

export const bundledLoader: BundledLoader = (ropes) => {
  const load = files[`./bundled/ready-${ropes}.json`];
  return load ? load() : Promise.reject(new Error(`no bundled board for ${ropes} ropes`));
};

export function fillFromBundled(stock: Stock, load: BundledLoader): Map<number, Promise<void>> {
  const done = new Map<number, Promise<void>>();
  for (const n of ROPE_COUNTS) {
    if (stock.has(n)) continue;
    done.set(n, load(n).then(
      (v) => {
        const b = decodeReady(v, n);
        if (b && !stock.has(n)) stock.put(b);
      },
      () => undefined,
    ));
  }
  return done;
}
```

```ts
export interface SavedGame {
  ropes: number;
  seed: number;
  moves: number;
  playedMs: number;
  snapshot: EngineSnapshot;
}

export function encodeGame(g: SavedGame): unknown {
  return { v: RECORD_VERSION, ropes: g.ropes, seed: g.seed, moves: g.moves, playedMs: g.playedMs, snapshot: encodeSnapshot(g.snapshot) };
}

export function decodeGame(v: unknown): SavedGame | null {
  if (!isObj(v) || v["v"] !== RECORD_VERSION) return null;
  const { ropes, seed, moves, playedMs } = v;
  if (!isInt(ropes) || !isInt(seed) || !isInt(moves) || moves < 0 || !isNum(playedMs) || playedMs < 0) return null;
  const snapshot = decodeSnapshot(v["snapshot"], ropes);
  if (!snapshot || snapshot.ropes.every((r) => "cleared" in r)) return null;
  return { ropes, seed, moves, playedMs, snapshot };
}
```

`src/game/saver.ts`:

```ts
import { fromSnapshot, toSnapshot } from "../engine/snapshot";
import { decodeGame, encodeGame, readRecord, removeRecord, writeRecord, type SavedGame } from "./records";
import { Session, type Phase } from "./session";
import type { KeyValueStore } from "./settings";

export interface CurrentGame {
  session: Session;
  ropes: number;
  seed: number;
  startedAt: number;
}

export class GameSaver {
  private game: CurrentGame | null = null;
  private lastPhase: Phase = "idle";
  private readonly store: KeyValueStore;
  private readonly key: string;
  private readonly now: () => number;

  constructor(store: KeyValueStore, key: string, now: () => number) {
    this.store = store;
    this.key = key;
    this.now = now;
  }

  load(): SavedGame | null {
    return readRecord(this.store, this.key, decodeGame);
  }

  started(g: CurrentGame): void {
    this.game = g;
    this.lastPhase = g.session.phase;
    this.write();
  }

  tick(): void {
    const g = this.game;
    if (!g) return;
    if (g.session.won) {
      removeRecord(this.store, this.key);
      this.game = null;
      return;
    }
    const phase = g.session.phase;
    if (phase === "idle" && this.lastPhase !== "idle") this.write();
    this.lastPhase = phase;
  }

  hidden(): void {
    if (this.game && this.game.session.phase === "idle" && !this.game.session.won) this.write();
  }

  forget(): void {
    this.game = null;
  }

  private write(): void {
    const g = this.game;
    if (!g) return;
    const s = g.session;
    writeRecord(this.store, this.key, encodeGame({
      ropes: g.ropes,
      seed: g.seed,
      moves: s.moves,
      playedMs: Math.round(this.now() - g.startedAt),
      snapshot: toSnapshot(s.engine),
    }));
  }
}

export function restoreGame(saved: SavedGame, now: number): { session: Session; startedAt: number } {
  const session = new Session(fromSnapshot(saved.snapshot));
  session.moves = saved.moves;
  return { session, startedAt: now - saved.playedMs };
}
```

`started` is called with a fresh session (`phase === "idle"`), so `toSnapshot` is safe there; `tick` and `hidden` only write at `"idle"`, where nothing is held, flying or landing.
